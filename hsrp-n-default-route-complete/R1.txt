config t

!
! Core Layer: R1 Router: ISR 4331 Router
!

no ip domain-lookup
hostname R1

! Console Settings
line con 0
 exec-timeout 5 0
 privilege level 15
 logging synchronous
 exit



! VTY (SSH)
line vty 0 4
 transport input ssh
 login local
 logging synchronous
 exit

! Security
username admin secret ccna123
enable secret ccna124
ip domain-name lab.local
ip ssh version 2
crypto key generate rsa general-keys modulus 2048









! Enable IPv6 Routing
ipv6 unicast-routing


! ========== STATIC ROUTES ==========

! Default routes to ISP
ip   route 0.0.0.0 0.0.0.0 203.0.113.1
ipv6 route ::/0 2001:db8:0:113::1

!z! Routes to R2's WAN
!zip   route 203.0.114.0 255.255.255.252 10.1.0.2
!zipv6 route 2001:db8:0:114::/64 2001:db8:0:1::2





! ========== LOOPBACK ==========

interface loopback0
 description R1 Loopback
 ip address 192.168.100.1 255.255.255.255
 ipv6 address 2001:db8:0:100::1/128
 no shutdown
 exit





! ========== WAN INTERFACES ==========

! R1 --- ISP          WAN to ISP
interface gig0/0/0
 description R1 --- ISP
 ip address 203.0.113.2              255.255.255.252
 ipv6 address 2001:db8:0:113::2/64
 ipv6 address fe80::1                link-local
 no shutdown
 exit


! R1 --- R2 via NIM-ES2-4 L2 Module
! First create VLAN 100 in vlan database

! Create SVI for VLAN 100
interface vlan 100
 description R1 --- R2 (Inter-router Link)
 ip address 10.1.0.1             255.255.255.252
 ipv6 address 2001:db8:0:1::1/64
 ipv6 address fe80::1            link-local
 no shutdown
 exit

! Configure physical NIM port
interface gig0/1/0
 description R1 --- R2
 switchport mode access
 switchport access vlan 100
 no shutdown
 exit





! ========== LAN INTERFACES (To Distribution Switches) ==========

! R1 --- SW1        L2 Trunk to SW1
interface gig0/0/1
 description R1 --- SW1 (Trunk)
 no ip address
 no ipv6 address
 no shutdown
 exit


! Subinterfaces for VLANs
interface gig0/0/1.10
 description R1 -- SW1 VLAN 10 (Eng)

! ========== HSRP CONFIGURATION ==========


! R1 interface gig0/0/1 --- SW1
! VLAN 10 (Eng) -- R1 is ACTIVE (Priority 110)
interface gig0/0/1.10
 standby version 2
 standby 10 ip 192.168.10.254
 ipv6 address autoconfig
 standby 10 ipv6 autoconfig
 standby 10 priority 110
 standby 10 preempt
 exit

! VLAN 20 (Sales) -- R1 is STANDBY (Default Priority 100)
interface gig0/0/1.20
 standby version 2
 standby 20 ip 192.168.20.254
 ipv6 address autoconfig
 standby 20 ipv6 autoconfig
 standby 20 preempt
 exit

! VLAN 30 (Admin) -- R1 is ACTIVE (Priority 110)
