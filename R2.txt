config t

!
! Core Layer: R2 Router: ISR 4331 Router
!

no ip domain-lookup
hostname R2

! Console Settings
line con 0
 exec-timeout 5 0
 privilege level 15
 logging synchronous
 exit



! VTY (SSH)
line vty 0 4
 transport input ssh
 login local
 logging synchronous
 exit

! Security
username admin secret ccna123
enable secret ccna124
ip domain-name lab.local
ip ssh version 2
crypto key generate rsa general-keys modulus 2048









! Enable IPv6 Routing
ipv6 unicast-routing


! ========== STATIC ROUTES ==========

! Default routes to ISP
ip   route 0.0.0.0 0.0.0.0 203.0.114.1
ipv6 route ::/0 2001:db8:0:114::1

! Routes to R1's WAN
ip   route 203.0.113.0 255.255.255.252 10.1.0.1
ipv6 route 2001:db8:0:113::/64 2001:db8:0:1::1





! ========== LOOPBACK ==========

interface loopback0
 description R2 Loopback
 ip address 192.168.1.2 255.255.255.255
 ipv6 address 2001:db8:0:1::2/128
 no shutdown
 exit





! ========== WAN INTERFACES ==========

! R2 --- ISP          WAN to ISP
interface gig0/0/0
 description R2 --- ISP
 ip address 203.0.114.2              255.255.255.252
 ipv6 address 2001:db8:0:114::2/64
 ipv6 address fe80::2                link-local
 no shutdown
 exit

!  R2 --- R1           WAN to R1 (for redundancy)
interface gig0/1/0
 description R2 --- R1
 ip   address 10.1.0.2           255.255.255.252
 ipv6 address 2001:db8:0:1::2/64
 ipv6 address fe80::2            link-local
 no shutdown
 exit

! ========== LAN INTERFACES (To Distribution Switches) ==========

! R2 --- SW1        L2 Trunk to SW1
interface gig0/0/1
 description R2 --- SW1 (Trunk)
 no ip address
 no ipv6 address
 no shutdown
 exit


! Subinterfaces for VLANs
interface gig0/0/1.10
 description R2 -- SW1 VLAN 10 (Eng)
 encapsulation dot1q 10
 ip   address 192.168.10.2        255.255.255.0
 ipv6 address 2001:db8:0:10::2/64
 ipv6 address fe80::210           link-local
 no shutdown
 exit

interface gig0/0/1.20
 description R2 -- SW1 VLAN 20 (Sales)
 encapsulation dot1q 20
 ip   address 192.168.20.2        255.255.255.0

 ipv6 address fe80::220           link-local
 no shutdown
 exit

interface gig0/0/1.30
 description R2 -- SW1 VLAN 30 (Admin)
 encapsulation dot1q 30
 ip address 192.168.30.2          255.255.255.0
 ipv6 address 2001:db8:0:30::2/64
 ipv6 address fe80::230           link-local
 no shutdown
 exit

interface gig0/0/1.99
 description R2 -- SW1 VLAN 99 (Management)
 encapsulation dot1q 99
 ip   address 192.168.99.2        255.255.255.0
 ipv6 address 2001:db8:0:99::2/64
 ipv6 address fe80::299           link-local
 no shutdown
 exit

 description R2 -- SW1 Native VLAN

 no shutdown
 exit
