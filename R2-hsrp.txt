!# configure terminal
! R2-hsrp.txt
! Part 1
!



! R2 interface gig0/0/1 --- SW1
! VLAN 10 (Eng) -- R2 is STANDBY (Default Priority 100)
interface gig0/0/1.10
 standby version 2
 standby 10 ip 192.168.10.254
! standby 10 ipv6 2001:db8:0:10::254
 ipv6 address autoconfig
 standby 10 ipv6 autoconfig
 standby 10 preempt
 exit

! VLAN 20 (Sales) -- R2 is ACTIVE (Priority 110)
interface gig0/0/1.20
 standby version 2
 standby 20 ip 192.168.20.254
! standby 20 ipv6 2001:db8:0:20::254
 ipv6 address autoconfig
 standby 20 ipv6 autoconfig
 standby 20 priority 220
 standby 20 preempt
 exit

! VLAN 30 (Admin) -- R2 is STANDBY (Default Priority 100)

! standby 30 ipv6 2001:db8:0:30::254

 exit

! VLAN 99 (Management) -- R2 is ACTIVE (Priority 110)
