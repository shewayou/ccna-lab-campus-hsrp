
!
! Rest assure in real world, these commands are entered in global config mode on ISR 4331 router.
! For now it is entered in privileged exec mode b/c Packet Tracer PT quirk on ISR 4331 with NIM-ES2-4 L2 port.
!
vlan database
vlan 10 name Eng
vlan 20 name Sales
vlan 30 name Admin
vlan 99 name Management
vlan 900 name Native
vlan 100 name L2-port-between-R1-R2
exit




config t

!
! Core Layer: R2 Router: ISR 4331 Router
!

no ip domain-lookup
hostname R2

! Console Settings
line con 0
 exec-timeout 5 0
 privilege level 15
 logging synchronous
 exit



! VTY (SSH)
line vty 0 4
 transport input ssh
 login local
 logging synchronous
 exit

! Security
username admin secret ccna123
enable secret ccna124
ip domain-name lab.local
ip ssh version 2
crypto key generate rsa general-keys modulus 2048









! Enable IPv6 Routing
ipv6 unicast-routing


! ========== STATIC ROUTES ==========

! Default routes to ISP
ip   route 0.0.0.0 0.0.0.0 203.0.114.1
ipv6 route ::/0 2001:db8:0:114::1







! ========== LOOPBACK ==========

interface loopback0
 description R2 Loopback
 ip address 192.168.100.2 255.255.255.255
 ipv6 address 2001:db8:0:100::2/128
 no shutdown
 exit





! ========== WAN INTERFACES ==========

! R2 --- ISP          WAN to ISP
interface gig0/0/0
 description R2 --- ISP
 ip address 203.0.114.2              255.255.255.252
 ipv6 address 2001:db8:0:114::2/64
 ipv6 address fe80::2                link-local
 no shutdown
 exit


! R1 --- R2 via NIM-ES2-4 L2 Module
! First create VLAN 100 in vlan database

! Create SVI for VLAN 100
interface vlan 100
 description R1 --- R2 (Inter-router Link)
 ip address 10.1.0.2             255.255.255.252
 ipv6 address 2001:db8:0:1::2/64
 ipv6 address fe80::2            link-local
 no shutdown
 exit

! Configure physical NIM port
interface gig0/1/0
 description R1 --- R2
 switchport mode access
 switchport access vlan 100
 no shutdown
 exit





! ========== LAN INTERFACES (To Distribution Switches) ==========

! R2 --- SW5        L2 Trunk to SW5
interface gig0/0/1
 description R2 --- SW5 (Trunk)
 no ip address
 no ipv6 address
 no shutdown
 exit


! Subinterfaces for VLANs
interface gig0/0/1.10
 description R2 -- SW5 VLAN 10 (Eng)
 encapsulation dot1q 10
 ip   address 192.168.10.2        255.255.255.0
 ipv6 address 2001:db8:0:10::2/64
 ipv6 address fe80::210           link-local
 no shutdown
 exit

interface gig0/0/1.20
 description R2 -- SW5 VLAN 20 (Sales)
 encapsulation dot1q 20
 ip   address 192.168.20.2        255.255.255.0
 ipv6 address 2001:db8:0:20::2/64
 ipv6 address fe80::220           link-local
 no shutdown
 exit

interface gig0/0/1.30
 description R2 -- SW5 VLAN 30 (Admin)
 encapsulation dot1q 30
 ip address 192.168.30.2          255.255.255.0
 ipv6 address 2001:db8:0:30::2/64
 ipv6 address fe80::230           link-local
 no shutdown
 exit

interface gig0/0/1.99
 description R2 -- SW5 VLAN 99 (Management)
 encapsulation dot1q 99
 ip   address 192.168.99.2        255.255.255.0
 ipv6 address 2001:db8:0:99::2/64
 ipv6 address fe80::299           link-local
 no shutdown
 exit

interface gig0/0/1.900
 description R2 -- SW5 Native VLAN
 encapsulation dot1q 900 native
 no shutdown
 exit








! ========== HSRP CONFIGURATION ==========

! R2 interface gig0/0/1 --- SW5
! VLAN 10 (Eng) -- R2 is STANDBY (Default Priority 100)
interface gig0/0/1.10
 standby version 2
 standby 10 ip 192.168.10.254
 ipv6 address autoconfig
 standby 10 ipv6 autoconfig
 standby 10 preempt
 exit

! VLAN 20 (Sales) -- R2 is ACTIVE (Priority 110)
interface gig0/0/1.20
 standby version 2
 standby 20 ip 192.168.20.254
 ipv6 address autoconfig
 standby 20 ipv6 autoconfig
 standby 20 priority 220
 standby 20 preempt
 exit

! VLAN 30 (Admin) -- R2 is STANDBY (Default Priority 100)
interface gig0/0/1.30
 standby version 2
 standby 30 ip 192.168.30.254
 ipv6 address autoconfig
 standby 30 ipv6 autoconfig
 standby 30 preempt
 exit

! VLAN 99 (Management) -- R2 is ACTIVE (Priority 110)
interface gig0/0/1.99
 standby version 2
 standby 99 ip 192.168.99.254
 ipv6 address autoconfig
 standby 99 ipv6 autoconfig
 standby 99 priority 110
 standby 99 preempt
 exit




! ========== DHCP POOLS ==========

! R2 can give out .101 --- .200 
ip dhcp excluded-address 192.168.10.1 192.168.10.100
ip dhcp excluded-address 192.168.10.201 192.168.10.254

ip dhcp excluded-address 192.168.20.1 192.168.20.100
ip dhcp excluded-address 192.168.20.201 192.168.20.254

ip dhcp excluded-address 192.168.30.1 192.168.30.100
ip dhcp excluded-address 192.168.30.201 192.168.30.254




ip dhcp pool Eng_VLAN10
 network 192.168.10.0 255.255.255.0
 default-router 192.168.10.254
 dns-server 8.8.8.8
 exit

ip dhcp pool Sales_VLAN20
 network 192.168.20.0 255.255.255.0
 default-router 192.168.20.254
 dns-server 8.8.8.8
 exit

ip dhcp pool Admin_VLAN30
 network 192.168.30.0 255.255.255.0
 default-router 192.168.30.254
 dns-server 8.8.8.8
 exit










! ========== OSPF CONFIGURATION ==========


router ospf 1
 router-id  2.2.2.2
 
 ! Best practice: Make all interfaces passive by default, then enable Hellos only VLAN 100
 ! VLAN 100 is L2-port-between-R1-R2

 ! passive-interface default
 passive-interface loopback0
 ! no passive-interface vlan 100
 passive-interface gig0/0/1.10
 passive-interface gig0/0/1.20
 passive-interface gig0/0/1.30
 passive-interface gig0/0/1.99
 

 ! Advertise Campus Networks into Area 0
 network 192.168.100.0  0.0.0.255    area 0
 network 10.1.0.0       0.0.0.3      area 0
 network 192.168.10.0   0.0.0.255    area 0
 network 192.168.20.0   0.0.0.255    area 0
 network 192.168.30.0   0.0.0.255    area 0
 network 192.168.99.0   0.0.0.255    area 0
 
 ! Share static Internet default route with R1
 default-information originate
 
 exit
 
 
! ================= IPv6 OSPF AREA 0 =================  
ipv6 router ospf 1
 router-id 2.2.2.2
 passive-interface loopback0
 ! no passive-interface vlan 100
 passive-interface gig0/0/1.10
 passive-interface gig0/0/1.20
 passive-interface gig0/0/1.30
 passive-interface gig0/0/1.99


 default-information originate
 
 exit
 
 
 
! Enable IPv6 OSPF directly on interfaces
interface loopback0
 ipv6 ospf 1  area 0
 exit

interface vlan 100
 ipv6 ospf 1  area 0
 exit

interface gig0/0/1.10
 ipv6 ospf 1  area 0
 exit

interface gig0/0/1.20
 ipv6 ospf 1  area 0
 exit

interface gig0/0/1.30
 ipv6 ospf 1  area 0
 exit

interface gig0/0/1.99
 ipv6 ospf 1  area 0
 exit


 
exit
