config t

!
! ISP Router - ISR4331
!

no ip domain-lookup
hostname ISP

! Console Settings
line con 0
 exec-timeout 5 0
 privilege level 15
 logging synchronous
 exit









! Enable IPv6
ipv6 unicast-routing


! Simulated Internet reachability
ip   route 192.168.0.0       255.255.0.0      203.0.113.2
ip   route 10.0.0.0          255.255.255.252    203.0.113.2
ipv6 route 2001:db8:0::/48                      2001:db8:0:113::2


! ADD redundant return routes pointing to R2:
ip   route 192.168.0.0       255.255.0.0        203.0.114.2
ip   route 10.0.0.0          255.255.255.252    203.0.114.2
ipv6 route 2001:db8:0::/48                      2001:db8:0:114::2



interface loopback0
 description Simulated Internet DNS
 ip address 8.8.8.8 255.255.255.255
 exit
 


! ISP --- R1    Interface to R1
interface gig0/0/0
 description ISP --- R1
 ip   address 203.0.113.1           255.255.255.252
 ipv6 address 2001:db8:0:113::1/64
 ipv6 address fe80::1000            link-local
 no shutdown
 exit

! ISP --- R2    Interface to R2
interface gig0/0/1
 description ISP --- R2
 ip   address 203.0.114.1           255.255.255.252
 ipv6 address 2001:db8:0:114::1/64
 ipv6 address fe80::1000            link-local
 no shutdown
 exit



exit
